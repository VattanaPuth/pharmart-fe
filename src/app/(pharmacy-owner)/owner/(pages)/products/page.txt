"use client";

import { useEffect, useState } from "react";
import Image from "next/image";
import { FiEdit2, FiTrash2, FiAlertTriangle } from "react-icons/fi";

import api from "@/lib/axios";

import DeleteModal from "../../components/products/Deletemodal";
import ProductModal from "../../components/products/PoductModal";

export default function ProductsPage() {
  const [items, setItems] = useState([]);
  const [categories, setCategories] = useState([]);

  const [loading, setLoading] = useState(true);

  const [selectedCategory, setSelectedCategory] = useState("all");

  const [deleteTargetId, setDeleteTargetId] = useState(null);

  const [modal, setModal] = useState({
    open: false,
    mode: "add",
    product: null,
  });

  // =========================================
  // FETCH CATEGORIES
  // =========================================
  const fetchCategories = async () => {
    try {
      const response = await api.get("/public/categories/read");

      setCategories(response.data || []);
    } catch (error) {
      console.error("Failed to fetch categories:", error);
    }
  };

  // =========================================
  // FETCH PRODUCTS
  // =========================================
  const fetchProducts = async (categoryId = selectedCategory) => {
    try {
      setLoading(true);

      const params = {};

      // FILTER CATEGORY
      if (categoryId !== "all") {
        params.category_id = categoryId;
      }

      const response = await api.get("/owner/products/read", {
        params,
      });

      setItems(response.data.data || []);
    } catch (error) {
      console.error("Failed to fetch products:", error);
    } finally {
      setLoading(false);
    }
  };

  // =========================================
  // INITIAL LOAD
  // =========================================
  useEffect(() => {
    fetchCategories();
  }, []);

  // =========================================
  // FILTER CHANGE
  // =========================================
  useEffect(() => {
    fetchProducts(selectedCategory);
  }, [selectedCategory]);

  // =========================================
  // DELETE
  // =========================================
  const handleDeleteClick = (id) => {
    setDeleteTargetId(id);
  };

  const handleConfirmDelete = async () => {
    try {
      await api.delete(`/owner/products/delete/${deleteTargetId}`);

      setItems((prev) => prev.filter((item) => item.id !== deleteTargetId));

      setDeleteTargetId(null);
    } catch (error) {
      console.error("Delete failed:", error);
    }
  };

  const handleCancelDelete = () => {
    setDeleteTargetId(null);
  };

  // =========================================
  // ADD PRODUCT
  // =========================================
  const handleAddClick = () => {
    setModal({
      open: true,
      mode: "add",
      product: null,
    });
  };

  // =========================================
  // EDIT PRODUCT
  // =========================================
  const handleEditClick = (product) => {
    setModal({
      open: true,
      mode: "edit",
      product: {
        ...product,

        // normalize for modal
        name: product.product_name,
        image: product.main_image,

        category: product.category?.id || "",
        category_name: product.category?.name,
        subcategory: product.subcategory_id,

        packages:
          product.packages?.map((p) => ({
            id: p.id,
            name: p.package_name,
            price: p.price,
            stock: p.stock_quantity,
            contains: p.contains,
            lowStockAlert: p.low_stock_threshold ?? 10,
            is_default: Boolean(p.is_default),
          })) || [],
      },
    });
  };
  // =========================================
  // SAVE PRODUCT
  // =========================================
  const handleModalSave = async () => {
    await fetchProducts();

    setModal({
      open: false,
      mode: "add",
      product: null,
    });
  };

  const handleModalClose = () => {
    setModal({
      open: false,
      mode: "add",
      product: null,
    });
  };

  return (
    <div className="bg-gray-50 w-full min-h-screen">
      {/* Delete Modal */}
      {deleteTargetId !== null && (
        <DeleteModal
          onConfirm={handleConfirmDelete}
          onCancel={handleCancelDelete}
        />
      )}

      {/* Product Modal */}
      <ProductModal
        setModal={setModal}
        isOpen={modal.open}
        mode={modal.mode}
        product={modal.product}
        onClose={handleModalClose}
        onSave={handleModalSave}
        categories={categories}
      />

      <div className="px-8 py-8">
        {/* Header */}
        <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-4 mb-6">
          <div>
            <h1 className="text-2xl font-bold text-slate-900">Products</h1>

            <p className="text-sm text-slate-400 mt-1">
              {items.length} active products
            </p>
          </div>

          <div className="flex items-center gap-3">
            {/* CATEGORY FILTER */}
            <select
              value={selectedCategory}
              onChange={(e) => setSelectedCategory(e.target.value)}
              className="px-4 py-2.5 rounded-xl border border-slate-200 bg-white text-sm text-slate-700 outline-none focus:ring-2 focus:ring-pink-400"
            >
              {/* ALL */}
              <option value="all">All Categories</option>

              {/* UNCATEGORIZED */}
              <option value="uncategorized">Uncategorized</option>

              {/* API CATEGORIES */}
              {categories
                .filter((category) => category.active)
                .map((category) => (
                  <option key={category.id} value={category.id}>
                    {category.name}
                  </option>
                ))}
            </select>
            {/* ADD BUTTON */}
            <button
              onClick={handleAddClick}
              className="flex items-center gap-2 bg-linear-to-r from-pink-500 to-rose-400 text-white text-sm font-semibold px-5 py-2.5 rounded-full shadow-sm hover:shadow-md hover:from-pink-600 hover:to-rose-500 transition-all duration-200"
            >
              <span className="text-lg leading-none">+</span>
              Add Product
            </button>
          </div>
        </div>

        {/* Loading */}
        {loading ? (
          <div className="bg-white rounded-2xl shadow-sm py-20 text-center">
            <div className="text-slate-400 text-sm">Loading products...</div>
          </div>
        ) : (
          <div className="bg-white rounded-2xl shadow-sm overflow-hidden">
            <table className="w-full border-collapse">
              <thead>
                <tr className="border-b border-slate-100">
                  {[
                    "Product",
                    "Category",
                    "Price",
                    "Stock",
                    "Expiry",
                    "Actions",
                  ].map((col) => (
                    <th
                      key={col}
                      className="px-6 py-4 text-left text-xs font-semibold text-slate-400 tracking-wide uppercase"
                    >
                      {col}
                    </th>
                  ))}
                </tr>
              </thead>

              <tbody>
                {items.map((product, i) => {
                  const defaultPackage = product.packages?.find(
                    (p) => p.is_default,
                  );

                  const stock = Number(defaultPackage?.stock_quantity ?? 0);
                  const threshold = Number(
                    defaultPackage?.low_stock_threshold ?? 10,
                  );

                  const isLowStock = stock > 0 && stock <= threshold;

                  return (
                    <tr
                      key={product.id}
                      className={`border-b border-slate-50 hover:bg-slate-50/70 transition-colors duration-150 ${
                        i === items.length - 1 ? "border-none" : ""
                      }`}
                    >
                      {/* Product */}
                      <td className="px-6 py-4">
                        <div className="flex items-center gap-3">
                          <div className="w-11 h-11 rounded-xl bg-slate-100 overflow-hidden shrink-0">
                            <img
                              src={
                                `${process.env.NEXT_PUBLIC_STORAGE_URL}/storage/${product.main_image}` ||
                                "/images/placeholder.png"
                              }
                              alt={product.product_name || "Product"}
                              width={44}
                              height={44}
                              className="object-cover w-full h-full"
                            />
                          </div>

                          <div>
                            <div className="text-sm font-semibold text-slate-800">
                              {product.product_name}
                            </div>

                            <div className="text-xs text-slate-400 mt-0.5">
                              {product.description || "No description"}
                            </div>
                          </div>
                        </div>
                      </td>

                      {/* Category */}
                      <td className="px-6 py-4 text-sm text-slate-600">
                        {product.category?.active === 0 || !product.category
                          ? "Uncategorized"
                          : product.category?.name}
                      </td>

                      {/* Price */}
                      <td className="px-6 py-4">
                        <div className="text-sm font-semibold text-slate-800">
                          ${product.price || 0}
                        </div>
                      </td>

                      {/* Stock */}
                      <td className="px-6 py-4">
                        {isLowStock ? (
                          <span className="inline-flex items-center gap-1.5 text-orange-500 text-sm font-medium">
                            <FiAlertTriangle size={14} />

                            {product.stock_quantity}
                          </span>
                        ) : (
                          <span className="text-sm text-slate-700">
                            {product.stock_quantity}
                          </span>
                        )}
                      </td>

                      {/* Expiry */}
                      <td className="px-6 py-4 text-sm text-slate-500">
                        {product.expiry_date || "-"}
                      </td>

                      {/* Actions */}
                      <td className="px-6 py-4">
                        <div className="flex items-center gap-3">
                          <button
                            onClick={() => handleEditClick(product)}
                            className="text-slate-400 hover:text-blue-500 transition-colors duration-150"
                            title="Edit product"
                          >
                            <FiEdit2 size={16} />
                          </button>

                          <button
                            onClick={() => handleDeleteClick(product.id)}
                            className="text-slate-400 hover:text-red-500 transition-colors duration-150"
                            title="Delete product"
                          >
                            <FiTrash2 size={16} />
                          </button>
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>

            {/* Empty State */}
            {items.length === 0 && (
              <div className="text-center py-16 text-slate-400">
                <div className="text-4xl mb-3">📦</div>

                <div className="text-sm font-medium">No products found</div>

                <div className="text-xs mt-1">Try changing category filter</div>
              </div>
            )}
          </div>
        )}
      </div>
    </div>
  );
}
